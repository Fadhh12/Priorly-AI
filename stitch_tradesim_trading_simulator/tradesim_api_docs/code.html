// Next.js / TypeScript App Router Implementation 
import { NextResponse } from 'next/server';

interface OrderPayload {
  symbol: string;
  side: 'BUY' | 'SELL';
  type: 'LIMIT' | 'MARKET';
  price: number;
  quantity: number;
  trader_id: string;
}

export async function POST(req: Request) {
  const body: OrderPayload = await req.json();
  const gatewayRes = await fetch('http://localhost:8000/api/v1/orders', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-KEY': process.env.TRADESIM_API_KEY!,
      'X-TRADESIM-SIGNATURE': process.env.TRADESIM_SECRET!
    },
    body: JSON.stringify(body)
  });

  const data = await gatewayRes.json();
  return NextResponse.json(data, { status: gatewayRes.status });
}
